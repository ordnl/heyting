\begin{definition}{(Terms in $\HA$)}
    We define the terms of our language recursively:
    \begin{itemize}
        \item $0$ is a term.
        \item If $t, s$ are terms, then $S(t), t+s, t\cdot s$ are terms.
    \end{itemize}
\end{definition}
\begin{definition}{(Numerals, closed terms, values)}
    \begin{itemize}
        \item A numeral of $n \in \mathbb{N}$ is $\overline{n} := \underbrace{S(\dots S(n)))}_{\text{$n$-times}}$
        \item A term is closed if it contains no free variables.
        \item The value of a closed term is defined recursively:
        \begin{itemize}
            \item $\val(0) = 0_\mathbb{N}$
            \item $\val(S(t)) = \val(t) +_\mathbb{N} 1_\mathbb{N}$
            \item $\val(t + s)) = \val(t) +_\mathbb{N} \val(s)$
            \item $\val(t \cdot s) = \val(t) \cdot_\mathbb{N} \val(s)$
        \end{itemize}
    \end{itemize}
\end{definition}
\begin{definition}{(Formula in $\HA$)}
    We define the formulae of our language recursively:
    \begin{itemize}
        \item $\bot$ is an atomic formula.
        \item If $t,s$ are terms, then $t=s$ is an atomic formulae.
        \item If $P,Q$ are formulae, then $P \land Q, P \lor Q, P \to Q, \forall x\; P, \exists x\; P$ are formulae
    \end{itemize}
    A formula without free variables is closed. \todo{Define free/bound variables properly later}
\end{definition}
\begin{definition}{(True and false atoms)}
    A closed atom $s = t$ is \emph{true} if $\val(s)=\val(t)$, and false otherwise. $\bot$ is a false atom.
\end{definition}

The main properties we need to prove for $\HA_\omega$:

\begin{lemma}{(Weakening)}
If $\seqd{\Gamma}{C}{\alpha}{r}$, then $\seqd{\Gamma,\Pi}{C}{\alpha}{r}$.
\end{lemma}

\begin{lemma}{(Inversion)}
    The following holds for some $\alpha$ and $r$:
    \begin{enumerate}
    \item $\seqd{A\wedge B,\Gamma}{C}{\alpha}{r}$ implies
      $\seqd{A,B,\Gamma}{C}{\alpha}{r}$;
    \item $\seqd{A\vee B,\Gamma}{C}{\alpha}{r}$ implies
      $\seqd{A,\Gamma}{C}{\alpha}{r}$ and $\seqd{B,\Gamma}{C}{\alpha}{r}$;
    \item $\seqd{A\rightarrow B,\Gamma}{C}{\alpha}{r}$ implies
      $\seqd{B,\Gamma}{C}{\alpha}{r}$;
    \item $\seqd{\exists x A(x),\Gamma}{C}{\alpha}{r}$ implies
      $\seqd{\Gamma,A(t)}{C}{\alpha}{r}$ for every $t$;
    \item $\seqd{\Gamma}{A\wedge B}{\alpha}{r}$ implies
      $\seqd{\Gamma}{A}{\alpha}{r}$ and $\seqd{\Gamma}{B}{\alpha}{r}$;
    \item $\seqd{\Gamma}{A\rightarrow B}{\alpha}{r}$ implies
      $\seqd{A,\Gamma}{B}{\alpha}{r}$;
    \item $\seqd{\Gamma}{\forall x A(x)}{\alpha}{r}$ implies
      $\seqd{\Gamma}{A(n)}{\alpha}{r}$ for every $t$.
    \end{enumerate}
    $(\lor R)$, $(\exists R)$, $(\forall L)$, and $(\to L)$ are not entirely invertible.
\end{lemma}

\subsection{Embedding}

To prove that $\HA$ embeds into $\HA_\omega$, we will want to prove induction admissibility:

\begin{lemma}{(Induction admissibility)}
    Given $\seqd{\Gamma}{A(0)}{\alpha}{r}$ and $\seqd{\Gamma,A(x)}{A(S(x))}{\beta}{r}$, then:
    \[ \seqd{\Gamma}{\forall x\; A(x)}{\gamma}{r}\]
\end{lemma}

However, this is not possible if our $\omega$-rule requires every term. As such we need to normalise our terms:

\begin{definition}{(Normalised terms)}
    For a closed formula $A$, we will call the normalised formula $A^*$, the result of replacing every closed term $t$ with $\val(t)$.
    \todo{I have to be careful with the notational and semantic stuff here}
\end{definition}
Evaluating all the closed terms will not change the truth value of a sequent. For free variables, we will effectively move the free variables out of the object language and consider their universal closure. As an example: 
\[ \seq{\Gamma}{a(a+b)=a^2+ab} \rightsquigarrow \forall m,n \in \mathbb{N} \; \left( \seq{\Gamma^*}{\bar{m}(\bar{m}+\bar{n})=\bar{m}^2-\bar{n}} \right) \]
Thus each sequent in this embedding will be mapped to multiple sequents. By making this accommodation, we find $\HA_\omega$ with only normalised formulae has admissibility of induction:

\begin{lemma}{(Induction admissibility)}
    Given $\seqd{\Gamma}{A(0)}{\alpha}{r}$ and $\seqd{\Gamma,A(n)}{A(S(n))}{\beta}{r}$, then:
    \[ \seqd{\Gamma}{\forall x\; A(x)}{\max(\alpha,\beta)+\omega}{r}\]
    where $\gr(A(n)) < r$
\end{lemma}
\begin{proof}
    This is done by combining instances of the premises with cut. As an example:
\begin{center}
\AxiomC{$\seqd{\Gamma}{A(0)}{\max(\alpha,\beta)+1}{r}$}
\AxiomC{$\seqd{\Gamma,A(0)}{A(1)}{\max(\alpha,\beta)+1}{r}$}
\RightLabel{$(\rm{Cut})$}
\BinaryInfC{$\seqd{\Gamma}{A(1)}{\max(\alpha,\beta)+1}{r}$}
\DisplayProof
\end{center}
    We can maintain $\gr(A(n)) < r$ and, we see the height increases by an extra inference. Thus in general we can prove $\seqd{\Gamma}{A(n)}{\max(\alpha,\beta) + n}{r}$. Therefore with the $\omega$-rule, we can obtain $\seqd{\Gamma}{\forall x\;A(x)}{\max(\alpha,\beta) + \omega}{r}$
\end{proof}

As we observe in induction admissibility, it will be the main contribution to an increase in height when translating a proof from $\HA$ to $\HA_\omega$. Thus each induction will add an extra $\omega$ to the height of the proof when converted to a $\HA_\omega$ proof:

\subsection{Rank reduction}
Before proving rank reduction, we need the following lemma:
\begin{lemma}{(Single-cut Rank Reduction)}
    Let $\gr(A) = r$, $\seqd{\Gamma}{A}{\alpha}{r}$ and $\seqd{\Pi,A}{C}{\beta}{r}$, then:
    \[ \seqd{\Gamma,\Pi}{C}{\omega \cdot (\alpha \# \beta) + \omega}{r} \]
\end{lemma}
For any cut-formula $A$ in a proof $\seqd{\Gamma}{C}{\alpha}{r}$ we require $\gr(A) < r$. Thus this lemma requires replacing the application of cut on $A$ to one on its subformulae. As an example, if $A = B \to B'$, suppose $A$ is principal formula on both sides of the cut. Therefore we have:
\begin{center}
    \AxiomC{$\seqd{\Gamma}{B}{\beta'}{r}$}
    \AxiomC{$\seqd{\Gamma,B'}{C}{\beta'}{r}$}
    \RightLabel{$(\to\rm{R})$}
    \BinaryInfC{$\seqd{\Gamma,B \to B'}{C}{\beta'}{r}$}
    \DisplayProof
    \quad
    \AxiomC{$\seqd{\Gamma,B}{B'}{\alpha'}{r}$}
    \RightLabel{$(\to\rm{L})$}
    \UnaryInfC{$\seqd{\Gamma}{B \to B'}{\alpha'}{r}$}
    \DisplayProof
\end{center}
We can therefore, apply cut on the premises:
\begin{center}
    \AxiomC{$\seqd{\Gamma}{B}{\gamma}{r}$}
    \AxiomC{$\seqd{\Gamma,B}{B'}{\gamma}{r}$}
    \AxiomC{$\seqd{\Gamma,B'}{C}{\gamma}{r}$}
    \RightLabel{$(\rm{Cut})$}
    \BinaryInfC{$\seqd{\Gamma,B}{C}{\gamma}{r}$}
    \RightLabel{$(\rm{Cut})$}
    \BinaryInfC{$\seqd{\Gamma}{C}{\gamma}{r}$}
    \DisplayProof
\end{center}
Therefore, we see this subproof contributes at most $\max(\alpha',\beta')+2$. If we have $n$ many $A = B \to B'$ in the proof, we can bound the whole proof with $\omega \cdot (\alpha \# \beta) + \omega$ (since $\max(\alpha,\beta) < \alpha \# \beta$). 

\begin{theorem}{(Rank reduction)}
    If $\seqd{\Gamma}{C}{\alpha}{r+1}$, then $\seqd{\Gamma}{C}{\omega^{\left(\omega^\alpha\right)}}{r}$
\end{theorem}
\begin{proof}

Let $A$ be the cut with $\gr(A) = r$, then we have premises:  $\seqd{\Gamma}{A}{\alpha_1}{r+1}$ and $\seqd{\Gamma,A}{C}{\alpha_2}{r+1}$ (where $\alpha_1,\alpha_2 < \alpha$). Applying the induction hypothesis on the two premises gives us: $\seqd{\Gamma}{A}{\omega^{\omega^\alpha_1}}{r}$ and $\seqd{\Gamma,A}{C}{\omega^{\omega^\alpha_2}}{r}$. We can apply our previous lemma on this to obtain: 
    \[ \seqd{\Gamma}{C}{\omega\cdot \left(\omega^{\omega^{\alpha_1}}\#\omega^{\omega^{\alpha_2}}\right) + \omega}{r} \]

\begin{theorem}{(Cut-elimination)}
    If $\seqd{\Gamma}{C}{\alpha}{r}$, then $\seqd{\Gamma}{C}{\omega_{2r}(\alpha)}{0}$
\end{theorem}
\begin{proof}
    By induction on $r$, an repeated application of Rank reduction. For the inductive step, we have $\seqd{\Gamma}{C}{\alpha}{r+1}$, then by Rank reduction $\seqd{\Gamma}{C}{\omega^{\left(\omega^\alpha\right)}}{r}$. This is equivalent to $\seqd{\Gamma}{C}{\omega_2(\alpha)}{r}$, which we apply our induction hypothesis will reduce to $\seqd{\Gamma}{C}{\omega_{2r}(\omega_2(\alpha))}{0}$. Finally this is equivalent to $\seqd{\Gamma}{C}{\omega_{2(r+1)}(\alpha)}{0}$
\end{proof}

\end{proof}
